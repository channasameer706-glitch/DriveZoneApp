          package com.example.drivezoneinspector.util
          import com.example.drivezoneinspector.model.DetectedFormat
          import com.example.drivezoneinspector.model.NumericField
          import org.json.JSONArray
          import org.json.JSONObject
          import org.xmlpull.v1.XmlPullParser
          import org.xmlpull.v1.XmlPullParserFactory
          import java.io.StringReader
          object FileParser {
              fun isBinary(bytes: ByteArray): Boolean {
                  var count = 0
                  val sample = minOf(bytes.size, 1024)
                  for (i in 0 until sample) {
                      val b = bytes[i].toInt()
                      if (b == 0) return true
                      if (b < 0x09 || (b in 0x0E..0x1F)) count++
                  }
                  return if (sample == 0) false else (count.toFloat() / sample.toFloat()) > 0.15f
              }
              fun detectFormat(text: String): DetectedFormat {
                  val t = text.trim()
                  if (t.isEmpty()) return DetectedFormat.PLAIN_TEXT
                  if ((t.startsWith("{") && t.endsWith("}")) || (t.startsWith("[") && t.endsWith("]"))) {
                      try {
                          if (t.startsWith("{")) JSONObject(t) else JSONArray(t)
                          return DetectedFormat.JSON
                      } catch (_: Exception) {}
                  }
                  if (t.startsWith("<") && t.endsWith(">")) {
                      try {
                          val f = XmlPullParserFactory.newInstance()
                          val p = f.newPullParser()
                          p.setInput(StringReader(t))
                          var e = p.eventType
                          var tags = 0
                          while (e != XmlPullParser.END_DOCUMENT) {
                              if (e == XmlPullParser.START_TAG) tags++
                              e = p.next()
                          }
                          if (tags > 0) return DetectedFormat.XML
                      } catch (_: Exception) {}
                  }
                  return DetectedFormat.PLAIN_TEXT
              }
              fun extractNumericFields(text: String, format: DetectedFormat): List<NumericField> {
                  val list = mutableListOf<NumericField>()
                  when (format) {
                      DetectedFormat.JSON -> {
                          try {
                              val t = text.trim()
                              if (t.startsWith("{")) parseJson(JSONObject(t), "root", list)
                          } catch (_: Exception) {}
                      }
                      DetectedFormat.XML, DetectedFormat.PLAIN_TEXT -> {
                          val r = Regex("""([a-zA-Z0-9_\.\-]+)\s*[:=]\s*(-?\d+(?:\.\d+)?)""")
                          r.findAll(text).forEach { m ->
                              val k = m.groupValues[1]
                              val s = m.groupValues[2]
                              val num = s.toDoubleOrNull()
                              if (num != null) list.add(NumericField(k, num, !s.contains(".")))
                          }
                      }
                      else -> {}
                  }
                  return list
              }
              private fun parseJson(obj: JSONObject, path: String, list: MutableList<NumericField>) {
                  val keys = obj.keys()
                  while (keys.hasNext()) {
                      val k = keys.next()
                      val v = obj.get(k)
                      val p = "$path.$k"
                      when (v) {
                          is Number -> list.add(NumericField(p, v.toDouble(), v is Int || v is Long))
                          is JSONObject -> parseJson(v, p, list)
                      }
                  }
              }
          }

          package com.example.drivezoneinspector.util
          import android.content.Context
          import android.net.Uri
          import com.example.drivezoneinspector.model.BackupItem
          import java.io.File
          import java.io.FileOutputStream
          import java.text.SimpleDateFormat
          import java.util.Date
          import java.util.Locale
          class BackupManager(private val context: Context) {
              private val folder: File get() = File(context.filesDir, "backups").apply { if (!exists()) mkdirs() }
              fun createBackup(uri: Uri, name: String): BackupItem? {
                  return try {
                      val ts = SimpleDateFormat("yyyyMMdd_HHmmss", Locale.getDefault()).format(Date())
                      val safe = name.replace(Regex("[^a-zA-Z0-9._-]"), "_")
                      val f = File(folder, "backup_${ts}_$safe")
                      context.contentResolver.openInputStream(uri)?.use { input ->
                          FileOutputStream(f).use { output -> input.copyTo(output) }
                      }
                      BackupItem(f.name, SimpleDateFormat("MMM dd, yyyy HH:mm:ss", Locale.getDefault()).format(Date()), f.absolutePath)
                  } catch (e: Exception) { null }
              }
              fun getBackups(): List<BackupItem> {
                  val files = folder.listFiles() ?: return emptyList()
                  return files.sortedByDescending { it.lastModified() }.map {
                      BackupItem(it.name, SimpleDateFormat("MMM dd, yyyy HH:mm:ss", Locale.getDefault()).format(Date(it.lastModified())), it.absolutePath)
                  }
              }
              fun restoreBackupToUri(path: String, dest: Uri): Boolean {
                  return try {
                      val src = File(path)
                      if (!src.exists()) return false
                      src.inputStream().use { input ->
                          context.contentResolver.openOutputStream(dest)?.use { output -> input.copyTo(output) }
                      }
                      true
                  } catch (e: Exception) { false }
              }
          }

          package com.example.drivezoneinspector.viewmodel
          import android.app.Application
          import android.net.Uri
          import android.provider.OpenableColumns
          import androidx.lifecycle.AndroidViewModel
          import androidx.lifecycle.viewModelScope
          import com.example.drivezoneinspector.model.*
          import com.example.drivezoneinspector.util.*
          import kotlinx.coroutines.Dispatchers
          import kotlinx.coroutines.flow.*
          import kotlinx.coroutines.launch
          import kotlinx.coroutines.withContext
          import java.io.ByteArrayOutputStream
          import java.util.Locale
          sealed class UIStatus {
              data object Idle : UIStatus()
              data object Loading : UIStatus()
              data class Success(val message: String) : UIStatus()
              data class Error(val message: String) : UIStatus()
          }
          class InspectorViewModel(application: Application) : AndroidViewModel(application) {
              private val backupManager = BackupManager(application)
              private val _currentFile = MutableStateFlow<InspectedFile?>(null)
              val currentFile: StateFlow<InspectedFile?> = _currentFile.asStateFlow()
              private val _status = MutableStateFlow<UIStatus>(UIStatus.Idle)
              val status: StateFlow<UIStatus> = _status.asStateFlow()
              private val _backups = MutableStateFlow<List<BackupItem>>(emptyList())
              val backups: StateFlow<List<BackupItem>> = _backups.asStateFlow()
              init { loadBackups() }
              fun loadBackups() {
                  viewModelScope.launch(Dispatchers.IO) { _backups.value = backupManager.getBackups() }
              }
              fun processSelectedFile(uri: Uri) {
                  viewModelScope.launch {
                      _status.value = UIStatus.Loading
                      try {
                          val ctx = getApplication<Application>()
                          var name = "save_file"
                          var size = 0L
                          ctx.contentResolver.query(uri, null, null, null, null)?.use { c ->
                              val nIdx = c.getColumnIndex(OpenableColumns.DISPLAY_NAME)
                              val sIdx = c.getColumnIndex(OpenableColumns.SIZE)
                              if (c.moveToFirst()) {
                                  if (nIdx != -1) name = c.getString(nIdx) ?: name
                                  if (sIdx != -1) size = c.getLong(sIdx)
                              }
                          }
                          withContext(Dispatchers.IO) { backupManager.createBackup(uri, name) }
                          loadBackups()
                          val bytes = withContext(Dispatchers.IO) {
                              val out = ByteArrayOutputStream()
                              ctx.contentResolver.openInputStream(uri)?.use { inp -> inp.copyTo(out) }
                              out.toByteArray()
                          }
                          if (FileParser.isBinary(bytes)) {
                              _currentFile.value = InspectedFile(uri, name, "${bytes.size} B", DetectedFormat.BINARY_OR_ENCRYPTED, null, emptyList(), false)
                          } else {
                              val text = String(bytes, Charsets.UTF_8)
                              val fmt = FileParser.detectFormat(text)
                              val nums = FileParser.extractNumericFields(text, fmt)
                              _currentFile.value = InspectedFile(uri, name, "${bytes.size} B", fmt, text.take(5000), nums, true)
                          }
                          _status.value = UIStatus.Success("Loaded safely. Backup saved.")
                      } catch (e: Exception) {
                          _status.value = UIStatus.Error("Error: ${e.localizedMessage}")
                      }
                  }
              }
              fun exportModifiedCopy(dest: Uri, field: NumericField, newVal: Double) {
                  viewModelScope.launch(Dispatchers.IO) {
                      _status.value = UIStatus.Loading
                      try {
                          val text = _currentFile.value?.rawPreviewText ?: return@launch
                          val p = Regex("(${Regex.escape(field.keyPath)}\\s*[:=]\\s*)(-?\\d+(?:\\.\\d+)?)")
                          val vStr = if (field.isInteger) newVal.toLong().toString() else newVal.toString()
                          val replaced = p.replaceFirst(text, "$1$vStr")
                          getApplication<Application>().contentResolver.openOutputStream(dest)?.use {
                              it.write(replaced.toByteArray(Charsets.UTF_8))
                          }
                          _status.value = UIStatus.Success("Exported to new copy safely.")
                      } catch (e: Exception) {
                          _status.value = UIStatus.Error("Export failed: ${e.localizedMessage}")
                      }
                  }
              }
              fun restoreBackupToExternal(item: BackupItem, dest: Uri) {
                  viewModelScope.launch(Dispatchers.IO) {
                      if (backupManager.restoreBackupToUri(item.absolutePath, dest)) {
                          _status.value = UIStatus.Success("Backup restored.")
                      } else {
                          _status.value = UIStatus.Error("Restore failed.")
                      }
                  }
              }
              fun resetStatus() { _status.value = UIStatus.Idle }
          }
